    private StackPane createUploadCard() {

        emptyUploadView =
                createEmptyUploadView();

        selectedImageView =
                createSelectedImageView();

        selectedImageView.setVisible(false);
        selectedImageView.setManaged(false);

        StackPane card =
                new StackPane(
                        emptyUploadView,
                        selectedImageView
                );

        card.setMinHeight(220);
        card.setPrefHeight(220);

        card.setPadding(
                new Insets(18)
        );

        card.setStyle(
                "-fx-background-color: white;"
                        + "-fx-background-radius: 14;"
                        + "-fx-border-color: #D7E3DE;"
                        + "-fx-border-radius: 14;"
        );

        card.setOnDragOver(
                event -> {

                    Dragboard board =
                            event.getDragboard();

                    if (
                            board.hasFiles()
                                    && isValidImage(
                                    board.getFiles()
                                            .get(0)
                            )
                    ) {

                        event.acceptTransferModes(
                                TransferMode.COPY
                        );
                    }

                    event.consume();
                }
        );

        card.setOnDragDropped(
                event -> {

                    Dragboard board =
                            event.getDragboard();

                    boolean success = false;

                    if (
                            board.hasFiles()
                                    && isValidImage(
                                    board.getFiles()
                                            .get(0)
                            )
                    ) {

                        loadImage(
                                board.getFiles()
                                        .get(0)
                        );

                        success = true;
                    }

                    event.setDropCompleted(
                            success
                    );

                    event.consume();
                }
        );

        return card;
    }

    // =========================================================
    // ANALYZE REAL IMAGE
    // =========================================================

    private void analyzeSelectedImage() {

        if (selectedImage == null) {
            return;
        }

        File imageToAnalyze =
                selectedImage;

        setProcessingState(true);

        Task<InferenceResult> task =
                new Task<>() {

                    @Override
                    protected InferenceResult call()
                            throws Exception {

                        return inferenceService
                                .analyze(
                                        imageToAnalyze
                                );
                    }
                };


        task.setOnSucceeded(
                event -> {

                    setProcessingState(false);

                    InferenceResult result =
                            task.getValue();

                    displayInferenceResult(
                            result
                    );
                }
        );


        task.setOnFailed(
                event -> {

                    setProcessingState(false);

                    Throwable error =
                            task.getException();

                    setErrorStatus();

                    showInformation(
                            "Analysis Failed",
                            error == null
                                    ? "Unknown inference error."
                                    : error.getMessage()
                    );
                }
        );


        Thread thread =
                new Thread(
                        task,
                        "ecosort-inference"
                );

        thread.setDaemon(true);
        thread.start();
    }


    // =========================================================
    // DISPLAY REAL RESULTS
    // =========================================================

    private void displayInferenceResult(
            InferenceResult result) {

        materialValue.setText(
                result.getHighestMaterial()
        );

        confidenceValue.setText(
                String.format(
                        "%.1f%%",
                        result.getHighestConfidence()
                                * 100.0
                )
        );


        // We do not fabricate segmentation metrics.
        purityValue.setText(
                "Pending"
        );

        contaminationValue.setText(
                "Pending"
        );


        // =============================================
        // Annotated image
        // =============================================

        File annotated =
                new File(
                        result.getAnnotatedImagePath()
                );

        if (annotated.exists()) {

            Image image =
                    new Image(
                            annotated.toURI()
                                    .toString(),
                            false
                    );

            imagePreview.setImage(
                    image
            );
        }


        // =============================================
        // Detection list
        // =============================================

        if (result.getDetectionCount() == 0) {

            detectionSummary.setText(
                    "No recyclable material was "
                            + "detected above the "
                            + "confidence threshold."
            );

            recommendationText.setText(
                    "UNKNOWN / MANUAL REVIEW: "
                            + "EcoSort could not identify "
                            + "a supported material reliably."
            );

        } else {

            StringBuilder detections =
                    new StringBuilder();


            for (Detection detection
                    : result.getDetections()) {

                if (detections.length() > 0) {
                    detections.append("\n");
                }

                detections.append(
                        "• "
                );

                detections.append(
                        detection.getMaterial()
                );

                detections.append(
                        "  —  "
                );

                detections.append(
                        String.format(
                                "%.1f%%",
                                detection.getConfidence()
                                        * 100.0
                        )
                );
            }


            detectionSummary.setText(
                    detections.toString()
            );


            updateRecommendation(
                    result
            );
        }


        compositionText.setText(
                "PLASTIC: "
                        + result.getPlasticCount()
                        + " detected\n"
                        + "METAL: "
                        + result.getMetalCount()
                        + " detected\n"
                        + "GLASS: "
                        + result.getGlassCount()
                        + " detected\n"
                        + "PAPER/CARDBOARD: "
                        + result.getPaperCardboardCount()
                        + " detected\n\n"
                        + "Note: object counts are shown "
                        + "until segmentation-mask area "
                        + "analysis is connected."
        );


        inferenceTimeLabel.setText(
                String.format(
                        "Current pipeline time: %.2f s "
                                + "(includes model startup)",
                        result.getInferenceTimeMs()
                                / 1000.0
                )
        );


        setCompleteStatus();
    }


    // =========================================================
    // RECOMMENDATION FROM REAL DETECTION CONFIDENCE
    // =========================================================

    private void updateRecommendation(
            InferenceResult result) {

        double confidence =
                result.getHighestConfidence();


        if (confidence < 0.40) {

            recommendationText.setText(
                    "UNKNOWN / MANUAL REVIEW: "
                            + "The highest detection confidence "
                            + "is too low for reliable recycling "
                            + "guidance."
            );

            return;
        }


        if (confidence < 0.60) {

            recommendationText.setText(
                    "LOW-CONFIDENCE RESULT: Possible "
                            + result.getHighestMaterial()
                            + " detected. Verify the material "
                            + "manually before recycling."
            );

            return;
        }


        recommendationText.setText(
                result.getHighestMaterial()
                        + " is the strongest detected material. "
                        + "Detection confidence is "
                        + String.format(
                        "%.1f%%",
                        confidence * 100.0
                )
                        + ". Final purity and contamination "
                        + "guidance will be calculated from "
                        + "segmentation-mask area."
        );
    }

    // =========================================================
    // IMAGE HANDLING
    // =========================================================

    private void chooseImage() {

        FileChooser chooser =
                new FileChooser();

        chooser.setTitle(
                "Select Waste Image"
        );

        chooser.getExtensionFilters()
                .add(
                        new FileChooser.ExtensionFilter(
                                "Image Files",
                                "*.jpg",
                                "*.jpeg",
                                "*.png"
                        )
                );

        File file =
                chooser.showOpenDialog(
                        stage
                );

        if (file != null) {
            loadImage(file);
        }
    }


    private void loadImage(
            File file) {

        if (!isValidImage(file)) {

            showInformation(
                    "Unsupported File",
                    "Please select a JPG, JPEG "
                            + "or PNG image."
            );

            return;
        }


        Image image =
                new Image(
                        file.toURI()
                                .toString(),
                        false
                );


        if (image.isError()) {

            showInformation(
                    "Image Error",
                    "EcoSort could not open "
                            + "this image."
            );

            return;
        }


        selectedImage =
                file;

        imagePreview.setImage(
                image
        );

        selectedFileName.setText(
                file.getName()
        );

        selectedFileDetails.setText(
                String.format(
                        "%.0f × %.0f px   •   %.1f KB",
                        image.getWidth(),
                        image.getHeight(),
                        file.length() / 1024.0
                )
        );


        emptyUploadView.setVisible(false);
        emptyUploadView.setManaged(false);

        selectedImageView.setVisible(true);
        selectedImageView.setManaged(true);

        resetResults();
        setReadyStatus();
    }


    private void clearImage() {

        selectedImage = null;

        imagePreview.setImage(null);

        selectedImageView.setVisible(false);
        selectedImageView.setManaged(false);

        emptyUploadView.setVisible(true);
        emptyUploadView.setManaged(true);

        resetResults();
        setReadyStatus();
    }


    private boolean isValidImage(
            File file) {

        if (
                file == null
                        || !file.isFile()
        ) {
            return false;
        }

        String name =
                file.getName()
                        .toLowerCase();

        return name.endsWith(".jpg")
                || name.endsWith(".jpeg")
                || name.endsWith(".png");
    }


    private void resetResults() {

        materialValue.setText("—");
        purityValue.setText("—");
        contaminationValue.setText("—");
        confidenceValue.setText("—");

        detectionSummary.setText(
                "No image has been analyzed yet."
        );

        inferenceTimeLabel.setText("");

        compositionText.setText(
                "Analyze an image to view "
                        + "detected material counts."
        );

        recommendationText.setText(
                "Upload and analyze an image "
                        + "to receive guidance."
        );
    }
